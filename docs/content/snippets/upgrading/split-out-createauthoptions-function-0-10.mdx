### Split out `createAuthOptions` function

Move your Better Auth options into a separate `createAuthOptions` function that
returns them as a typed object, and pass its result to `betterAuth()`. With the
options inside the `betterAuth()` call, the component had to run Better Auth to
read them, which caused unnecessary logs and inaccurate error messages, and
required workarounds such as `optionsOnly`.

```ts title="convex/auth.ts"
import {
  betterAuth,
  type BetterAuthOptions, // [!code ++]
} from "better-auth/minimal";

// [!code ++:5]
export const createAuthOptions = (ctx: GenericCtx<DataModel>) => {
  return {
    // ... auth config
  } satisfies BetterAuthOptions;
};

export const createAuth = (ctx: GenericCtx<DataModel>) => {
  // [!code --:3]
  return betterAuth({
    // ... auth config
  });
  return betterAuth(createAuthOptions(ctx)); // [!code ++]
};
```
