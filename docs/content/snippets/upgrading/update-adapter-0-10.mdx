### Update adapter

In `adapter.ts`, pass `createAuthOptions` to `createApi` instead of
`createAuth`, remove the `migrationRemoveUserId` export, which the component no
longer provides, and export the `consumeOne` and `incrementOne` functions that
Better Auth 1.7 adds.

```ts title="convex/betterAuth/adapter.ts"
import { createApi } from "@convex-dev/better-auth";
import {
  createAuth, // [!code --]
  createAuthOptions, // [!code ++]
} from "../auth";
import schema from "./schema";

export const {
  create,
  findOne,
  findMany,
  updateOne,
  updateMany,
  deleteOne,
  deleteMany,
  migrationRemoveUserId, // [!code --]
  consumeOne, // [!code ++]
  incrementOne, // [!code ++]
} = createApi(schema, createAuth); // [!code --]
} = createApi(schema, createAuthOptions); // [!code ++]
```
